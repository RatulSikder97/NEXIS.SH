# Phase 3.5 — Workspaces + Billing — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development.

**Goal:** Ship workspace onboarding (with provisioning animation) + Stripe-style mocked billing (payment method, invoices, AWS-style runtime metering) before Phase 4.

**Architecture:** Workspace = new compute unit inside an org. After signup, owner is bounced to `/onboarding/workspace`; console is unreachable until a workspace is ready. Provisioning state machine runs server-side (in-process goroutine), publishes events to an SSE broker; web consumes via EventSource and renders a 5-step progress bar. Billing is a port/adapter pair (LocalBillingProvider for Phase 3.5, StripeProvider stub for Phase 7). Cron records `usage_records` for ready workspaces.

**Tech Stack:** Go 1.25 control-plane, Next.js 16.2.2 web, Postgres 16 RLS, in-process SSE broker, `EventSource` browser API.

**Spec:** `docs/superpowers/specs/2026-05-13-phase-3.5-workspaces-billing.md`.

### Task 1.2: SSE broker

**Files:**
- Create: `services/control-plane/internal/platform/sse/broker.go`

```go
// Package sse is a tiny in-process pub/sub for Server-Sent Events.
// Topics are arbitrary strings (we use workspace id). Subscribers get
// a buffered channel; non-blocking publish drops events on slow consumers.
package sse

import "sync"

type Broker[T any] struct {
    mu   sync.RWMutex
    subs map[string]map[chan T]struct{}
}

func New[T any]() *Broker[T] {
    return &Broker[T]{subs: map[string]map[chan T]struct{}{}}
}

func (b *Broker[T]) Subscribe(topic string, buf int) (<-chan T, func()) {
    ch := make(chan T, buf)
    b.mu.Lock()
    if _, ok := b.subs[topic]; !ok { b.subs[topic] = map[chan T]struct{}{} }
    b.subs[topic][ch] = struct{}{}
    b.mu.Unlock()
    return ch, func() {
        b.mu.Lock()
        delete(b.subs[topic], ch)
        if len(b.subs[topic]) == 0 { delete(b.subs, topic) }
        b.mu.Unlock()
        close(ch)
    }
}

func (b *Broker[T]) Publish(topic string, ev T) {
    b.mu.RLock()
    defer b.mu.RUnlock()
    for ch := range b.subs[topic] {
        select { case ch <- ev: default: /* drop on slow consumer */ }
    }
}
```

package repo

import (
    "context"
    "errors"
    "time"

    "github.com/jackc/pgx/v5"
    "github.com/jackc/pgx/v5/pgxpool"

    "github.com/nexis-eco/nexis/services/control-plane/internal/domain"
    "github.com/nexis-eco/nexis/services/control-plane/internal/platform/db"
)

type WorkspacesRepo struct{ pool *pgxpool.Pool }

func NewWorkspacesRepo(pool *pgxpool.Pool) *WorkspacesRepo { return &WorkspacesRepo{pool: pool} }

func (r *WorkspacesRepo) Insert(ctx context.Context, w *domain.Workspace) error {
    q := db.FromCtx(ctx, r.pool)
    return q.QueryRow(ctx, `
        INSERT INTO workspaces (org_id, name, slug, region, status, provisioning_step, created_at, updated_at)
        VALUES ($1,$2,$3,$4,$5,$6, now(), now())
        RETURNING id, created_at, updated_at`,
        w.OrgID, w.Name, w.Slug, w.Region, string(w.Status), w.ProvisioningStep,
    ).Scan(&w.ID, &w.CreatedAt, &w.UpdatedAt)
}

func (r *WorkspacesRepo) UpdateStatus(ctx context.Context, id string, status domain.WorkspaceStatus, step, message string) error {
    q := db.FromCtx(ctx, r.pool)
    var readyAt *time.Time
    if status == domain.WSReady { n := time.Now(); readyAt = &n }
    _, err := q.Exec(ctx, `
        UPDATE workspaces
        SET status=$2, provisioning_step=$3, status_message=$4,
            ready_at = COALESCE($5, ready_at), updated_at=now()
        WHERE id=$1`, id, string(status), step, message, readyAt)
    return err
}

func (r *WorkspacesRepo) Get(ctx context.Context, orgID, id string) (*domain.Workspace, error) {
    q := db.FromCtx(ctx, r.pool)
    var w domain.Workspace
    err := q.QueryRow(ctx, `
        SELECT id, org_id, name, slug, region, status, COALESCE(status_message, ''), COALESCE(provisioning_step, ''), created_at, ready_at, updated_at
        FROM workspaces WHERE org_id=$1 AND id=$2`, orgID, id).Scan(
        &w.ID, &w.OrgID, &w.Name, &w.Slug, &w.Region, &w.Status, &w.StatusMessage, &w.ProvisioningStep, &w.CreatedAt, &w.ReadyAt, &w.UpdatedAt,
    )
    if errors.Is(err, pgx.ErrNoRows) { return nil, domain.ErrNotFound }
    return &w, err
}

func (r *WorkspacesRepo) List(ctx context.Context, orgID string) ([]domain.Workspace, error) {
    q := db.FromCtx(ctx, r.pool)
    rows, err := q.Query(ctx, `
        SELECT id, org_id, name, slug, region, status, COALESCE(status_message, ''), COALESCE(provisioning_step, ''), created_at, ready_at, updated_at
        FROM workspaces WHERE org_id=$1 ORDER BY created_at DESC`, orgID)
    if err != nil { return nil, err }
    defer rows.Close()
    out := []domain.Workspace{}
    for rows.Next() {
        var w domain.Workspace
        if err := rows.Scan(&w.ID, &w.OrgID, &w.Name, &w.Slug, &w.Region, &w.Status, &w.StatusMessage, &w.ProvisioningStep, &w.CreatedAt, &w.ReadyAt, &w.UpdatedAt); err != nil { return nil, err }
        out = append(out, w)
    }
    return out, nil
}

func (r *WorkspacesRepo) ListAllReady(ctx context.Context) ([]domain.Workspace, error) {
    // Used by the usage cron — bypasses RLS via the admin pool.
    rows, err := r.pool.Query(ctx, `
        SELECT id, org_id FROM workspaces WHERE status='ready'`)
    if err != nil { return nil, err }
    defer rows.Close()
    out := []domain.Workspace{}
    for rows.Next() {
        var w domain.Workspace
        if err := rows.Scan(&w.ID, &w.OrgID); err != nil { return nil, err }
        out = append(out, w)
    }
    return out, nil
}

func (r *WorkspacesRepo) HasAny(ctx context.Context, orgID string) (bool, error) {
    q := db.FromCtx(ctx, r.pool)
    var n int
    if err := q.QueryRow(ctx, `SELECT count(*) FROM workspaces WHERE org_id=$1`, orgID).Scan(&n); err != nil {
        return false, err
    }
    return n > 0, nil
}

```go
package workspace

import (
    "context"
    "fmt"
    "math/rand/v2"
    "strings"
    "time"

    "github.com/nexis-eco/nexis/services/control-plane/internal/adapter/repo"
    "github.com/nexis-eco/nexis/services/control-plane/internal/domain"
    "github.com/nexis-eco/nexis/services/control-plane/internal/platform/sse"
)

type Config struct {
    Repo        *repo.WorkspacesRepo
    Broker      *sse.Broker[domain.ProvisioningStep]
    FailRate    float64       // 0..1 — for demo error states
    StepBudget  time.Duration // total provisioning time (default 5s)
}

type Service struct {
    cfg Config
}

func New(cfg Config) *Service {
    if cfg.StepBudget == 0 { cfg.StepBudget = 5 * time.Second }
    return &Service{cfg: cfg}
}

var phases = []struct {
    step, label string
    fraction    float64 // share of total budget
}{
    {"creating_organization", "Creating organization…",   0.15},
    {"allocating_host",        "Allocating host…",         0.25},
    {"provisioning_datacenter","Provisioning datacenter…", 0.30},
    {"deploying",              "Deploying workspace…",     0.30},
}

func (s *Service) Create(ctx context.Context, p domain.Principal, name, region string) (domain.Workspace, error) {
    if !regionAllowed(region) { return domain.Workspace{}, fmt.Errorf("unknown region %q", region) }
    slug := slugify(name)
    w := &domain.Workspace{
        OrgID: p.OrgID, Name: name, Slug: slug, Region: region,
        Status: domain.WSProvisioning, ProvisioningStep: phases[0].step,
    }
    if err := s.cfg.Repo.Insert(ctx, w); err != nil { return domain.Workspace{}, err }

    // Drive the state machine on a background goroutine. Use a fresh ctx so the
    // request being cancelled doesn't abort provisioning.
    go s.run(*w)

    return *w, nil
}

func (s *Service) run(w domain.Workspace) {
    ctx := context.Background()
    cumulative := 0.0
    willFail := rand.Float64() < s.cfg.FailRate

    for i, phase := range phases {
        dur := time.Duration(float64(s.cfg.StepBudget) * phase.fraction)
        time.Sleep(dur)
        cumulative += phase.fraction
        msg := ""
        status := "in_progress"
        if willFail && i == len(phases)/2 {
            status = "error"
            msg = "synthetic failure (demo)"
            s.cfg.Broker.Publish(w.ID, domain.ProvisioningStep{
                Step: phase.step, Label: phase.label, Progress: cumulative,
                Status: status, Message: msg, TS: time.Now(),
            })
            _ = s.cfg.Repo.UpdateStatus(ctx, w.ID, domain.WSError, phase.step, msg)
            return
        }
        _ = s.cfg.Repo.UpdateStatus(ctx, w.ID, domain.WSProvisioning, phase.step, "")
        s.cfg.Broker.Publish(w.ID, domain.ProvisioningStep{
            Step: phase.step, Label: phase.label, Progress: cumulative, Status: status, TS: time.Now(),
        })
    }
    // ready
    _ = s.cfg.Repo.UpdateStatus(ctx, w.ID, domain.WSReady, "ready", "")
    s.cfg.Broker.Publish(w.ID, domain.ProvisioningStep{
        Step: "ready", Label: "Workspace ready", Progress: 1.0, Status: "ready", TS: time.Now(),
    })
}

func (s *Service) Get(ctx context.Context, p domain.Principal, id string) (domain.Workspace, error) {
    w, err := s.cfg.Repo.Get(ctx, p.OrgID, id)
    if err != nil { return domain.Workspace{}, err }
    return *w, nil
}

func (s *Service) List(ctx context.Context, p domain.Principal) ([]domain.Workspace, error) {
    return s.cfg.Repo.List(ctx, p.OrgID)
}

func (s *Service) Suspend(ctx context.Context, p domain.Principal, id string) error {
    return s.cfg.Repo.UpdateStatus(ctx, id, domain.WSSuspended, "suspended", "")
}

func (s *Service) Events(ctx context.Context, workspaceID string) <-chan domain.ProvisioningStep {
    ch, unsub := s.cfg.Broker.Subscribe(workspaceID, 16)
    go func() { <-ctx.Done(); unsub() }()
    return ch
}

func regionAllowed(r string) bool {
    for _, x := range domain.Regions { if x.ID == r { return true } }
    return false
}

func slugify(s string) string {
    s = strings.ToLower(strings.TrimSpace(s))
    var b strings.Builder
    prevDash := false
    for _, c := range s {
        switch {
        case c >= 'a' && c <= 'z', c >= '0' && c <= '9':
            b.WriteRune(c); prevDash = false
        case !prevDash:
            b.WriteRune('-'); prevDash = true
        }
    }
    return strings.Trim(b.String(), "-")
}

package cron

import (
    "context"
    "log/slog"
    "time"
)

type Job struct {
    Name     string
    Interval time.Duration
    Run      func(ctx context.Context) error
}

func Start(ctx context.Context, logger *slog.Logger, jobs []Job) {
    for _, j := range jobs {
        j := j
        go func() {
            t := time.NewTicker(j.Interval)
            defer t.Stop()
            for {
                select {
                case <-ctx.Done(): return
                case <-t.C:
                    if err := j.Run(ctx); err != nil {
                        logger.Error("cron job", "name", j.Name, "err", err)
                    }
                }
            }
        }()
    }
}
```

package usecase

import (
    "context"
    "time"

    "github.com/nexis-eco/nexis/services/control-plane/internal/adapter/repo"
    "github.com/nexis-eco/nexis/services/control-plane/internal/domain"
)

type UsageTicker struct {
    Workspaces *repo.WorkspacesRepo
    Billing    *repo.BillingRepo
    Interval   time.Duration   // seconds between ticks (also fragment size)
    PriceCents float64         // $0.10/hr = 10.0
}

func (u *UsageTicker) Run(ctx context.Context) error {
    fraction := u.Interval.Hours() // tick interval in hours
    ws, err := u.Workspaces.ListAllReady(ctx)
    if err != nil { return err }
    for _, w := range ws {
        amount := fraction * u.PriceCents
        _ = u.Billing.RecordUsage(ctx, domain.UsageRecord{
            OrgID: w.OrgID, WorkspaceID: w.ID, Project: "default",
            Kind: "runtime_hours",
            Quantity: fraction, UnitPriceCents: u.PriceCents, AmountCents: amount,
            RecordedAt: time.Now().UTC(),
        })
    }
    return nil
}
